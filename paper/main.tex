\begin{abstract}
Punjab 44$^\circ$C Loo and monsoon humidity compress shelf-life via $\Phi_{env}$; we present a 4-stage optimizer (Arrhenius $\rightarrow$ demand LSTM/LightGBM $\rightarrow$ Pareto MILP $\rightarrow$ VRPTW OR-Tools) with Langar Rehat/Halal hard gates, evaluated on 23 districts (HVI) and NH44 corridor. Code and data: \texttt{data\_manifest.json} FAIR \cite{wilkinson2016fair}.
\end{abstract}

\section{Model}
\subsection{Shelf-life} $k(T)=A e^{-E_a/RT}$ \cite{arrhenius1889}, $\Phi_{env}=e^{\alpha(T-20)+\beta\max(0,H-60)}$, $t_{safe}=t_{base}/\Phi_{env}-t_{elapsed}$ \cite{labuza1993kinetics}, hazard $\le4h$ \cite{taoukis1997tti}.
\subsection{Matcher} $S_{ij}=w_1\text{Urgency}+w_2\text{Deficit}+w_3\text{Proximity}+w_4\text{Diet}$ $w=[0.35,0.30,0.20,0.15]$ \cite{saaty1980ahp}, MILP \cite{wolsey1998integer} via PuLP \cite{pulp2011} / CP-SAT \cite{orgtools2024}.
\subsection{VRP} $\min\sum c_{uw}x_{uvw}+\lambda\sum t_{delivery}/t_{safe}$ \cite{toth2014vrp} with tiers Tata Ace EV \cite{tataaceevspec} etc., OR-Tools \cite{orgtoolsvrp2024}, Solomon \cite{solomon1987} bench.
\subsection{Forecaster} HVI from NITI MPI \cite{niti2023mpi}, NFHS-5 \cite{nfhs5}, WFP \cite{wfp2024hungermap}, Census \cite{census2011}; LightGBM \cite{ke2017lightgbm}, LSTM \cite{hochreiter1997lstm}, TFT \cite{lim2021tft}.
